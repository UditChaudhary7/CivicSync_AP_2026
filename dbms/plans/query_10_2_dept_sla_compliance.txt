SELECT d.department_id, d.name AS department_name, COUNT(c.complaint_id) AS total_complaints, COUNT(c.complaint_id) FILTER (WHERE c.status IN ('RESOLVED', 'CLOSED')) AS resolved_count, COUNT(c.complaint_id) FILTER (WHERE c.status IN ('REPORTED', 'UNDER_REVIEW', 'ASSIGNED', 'IN_PROGRESS')) AS active_backlog, COUNT(c.complaint_id) FILTER (WHERE c.status NOT IN ('RESOLVED', 'CLOSED') AND CURRENT_TIMESTAMP > (c.created_at + (cat.sla_hours * INTERVAL '1 hour'))) AS sla_breached_count, ROUND(COALESCE(AVG(EXTRACT(EPOCH FROM (c.updated_at - c.created_at)) / 3600) FILTER (WHERE c.status IN ('RESOLVED', 'CLOSED')), 0)::NUMERIC, 2) AS avg_resolution_time_hours FROM departments d LEFT JOIN categories cat ON d.department_id = cat.department_id LEFT JOIN complaints c ON cat.category_id = c.category_id GROUP BY d.department_id, d.name ORDER BY total_complaints DESC;
