SELECT cat.category_id, cat.name AS category_name, d.name AS department_name, cat.default_priority, cat.sla_hours, COUNT(c.complaint_id) AS total_complaints, COUNT(c.complaint_id) FILTER (WHERE c.priority = 'CRITICAL') AS critical_count, COUNT(c.complaint_id) FILTER (WHERE c.priority = 'HIGH') AS high_count, COUNT(c.complaint_id) FILTER (WHERE c.priority = 'MEDIUM') AS medium_count, COUNT(c.complaint_id) FILTER (WHERE c.priority = 'LOW') AS low_count FROM categories cat INNER JOIN departments d ON cat.department_id = d.department_id LEFT JOIN complaints c ON cat.category_id = c.category_id GROUP BY cat.category_id, cat.name, d.name, cat.default_priority, cat.sla_hours ORDER BY total_complaints DESC;
