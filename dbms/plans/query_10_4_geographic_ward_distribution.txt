SELECT w.ward_id, w.ward_number, w.ward_name, w.zone_name, COUNT(c.complaint_id) AS total_complaints, COUNT(c.complaint_id) FILTER (WHERE c.status IN ('REPORTED', 'UNDER_REVIEW', 'ASSIGNED', 'IN_PROGRESS')) AS unresolved_complaints, COUNT(c.complaint_id) FILTER (WHERE c.status IN ('RESOLVED', 'CLOSED')) AS resolved_complaints FROM wards w LEFT JOIN complaints c ON w.ward_id = c.ward_id GROUP BY w.ward_id, w.ward_number, w.ward_name, w.zone_name ORDER BY total_complaints DESC;
