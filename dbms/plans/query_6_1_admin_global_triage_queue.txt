SELECT c.complaint_id, c.title, c.citizen_id, u.name AS citizen_name, cat.name AS category_name, d.name AS department_name, w.ward_number, w.zone_name, c.status, c.priority, c.created_at, c.created_at + (cat.sla_hours * INTERVAL '1 hour') AS sla_deadline, CASE WHEN c.status IN ('RESOLVED', 'CLOSED') THEN 'RESOLVED' WHEN CURRENT_TIMESTAMP > (c.created_at + (cat.sla_hours * INTERVAL '1 hour')) THEN 'SLA_BREACHED' ELSE 'WITHIN_SLA' END AS sla_status FROM complaints c INNER JOIN users u ON c.citizen_id = u.user_id INNER JOIN categories cat ON c.category_id = cat.category_id INNER JOIN departments d ON cat.department_id = d.department_id INNER JOIN wards w ON c.ward_id = w.ward_id WHERE c.status = 'IN_PROGRESS' AND c.category_id = 1 AND c.ward_id = 2 AND (c.title ILIKE '%pothole%' OR c.description ILIKE '%pothole%') ORDER BY CASE c.priority WHEN 'CRITICAL' THEN 1 WHEN 'HIGH' THEN 2 WHEN 'MEDIUM' THEN 3 WHEN 'LOW' THEN 4 ELSE 5 END ASC, c.created_at DESC LIMIT 25 OFFSET 0;
